class SimpleReflexVacuumAgent:
    def __init__(self):
        self.location = 'A'

    def sense_and_act(self, status):
        if status == 'Dirty':
            return 'Suck'
        elif self.location == 'A':
            return 'Right'
        elif self.location == 'B':
            return 'Left'

    def update_location(self, action):
        if action == 'Right':
            self.location = 'B'
        elif action == 'Left':
            self.location = 'A'


class VacuumEnvironment:
    def __init__(self, state_dict):
        self.states = state_dict

    def run_simulation(self, agent, steps=4):
        print(f"Initial Environment State: {self.states}")
        print(f"Agent starts at Location: {agent.location}\n")

        for step in range(1, steps + 1):
            current_status = self.states[agent.location]
            action = agent.sense_and_act(current_status)
            
            print(f"Step {step}: At {agent.location}, Status: {current_status} -> Action: {action}")
            
            if action == 'Suck':
                self.states[agent.location] = 'Clean'
            else:
                agent.update_location(action)
                
        print(f"\nFinal Environment State: {self.states}")


initial_state = {'A': 'Dirty', 'B': 'Dirty'}
my_agent = SimpleReflexVacuumAgent()
sim = VacuumEnvironment(initial_state)
sim.run_simulation(my_agent, steps=4)

import heapq

GOAL_STATE = (1, 2, 3, 
              4, 5, 6, 
              7, 8, 0)

def get_manhattan_distance(state):
    distance = 0
    for i in range(9):
        value = state[i]
        if value != 0: # Exclude the blank space from the calculation
            current_row, current_col = divmod(i, 3)
            target_idx = GOAL_STATE.index(value)
            target_row, target_col = divmod(target_idx, 3)
            
            distance += abs(current_row - target_row) + abs(current_col - target_col)
    return distance

def get_neighbors(state):
    neighbors = []
    blank_idx = state.index(0)
    row, col = divmod(blank_idx, 3)
    
    moves = {
        'Up': (-1, 0),
        'Down': (1, 0),
        'Left': (0, -1),
        'Right': (0, 1)
    }
    
    for direction, (dr, dc) in moves.items():
        new_row, new_col = row + dr, col + dc
        
        if 0 <= new_row < 3 and 0 <= new_col < 3:
            new_blank_idx = new_row * 3 + new_col
            
            state_list = list(state)
            state_list[blank_idx], state_list[new_blank_idx] = state_list[new_blank_idx], state_list[blank_idx]
            neighbors.append((tuple(state_list), direction))
            
    return neighbors

def solve_a_star(start_state):
    start_h = get_manhattan_distance(start_state)
    open_set = [(start_h, 0, start_state, [])]
    closed_set = set()
    
    while open_set:
        f, g, current, path = heapq.heappop(open_set)
        
        if current == GOAL_STATE:
            return path
            
        if current in closed_set:
            continue
            
        closed_set.add(current)
        
        for neighbor, direction in get_neighbors(current):
            if neighbor in closed_set:
                continue
                
            new_g = g + 1
            new_h = get_manhattan_distance(neighbor)
            new_f = new_g + new_h
            new_path = path + [(direction, neighbor)]
            
            heapq.heappush(open_set, (new_f, new_g, neighbor, new_path))
            
    return None 

def print_board(state):
    for i in range(0, 9, 3):
        print(f" {state[i]} {state[i+1]} {state[i+2]} ")
    print("-" * 11)

if __name__ == "__main__":
    initial_puzzle = (1, 2, 3, 
                      0, 4, 6,

    else:

ALPHA BETA PRUNING:


import math

def alpha_beta_pruning(depth, node_index, maximizing_player, values, alpha, beta):
    if depth == 0 or node_index >= len(values):
        return values[node_index]

    if maximizing_player:
        max_eval = -math.inf
        for i in range(2):  # Each node has two children
            eval = alpha_beta_pruning(depth - 1, node_index * 2 + i, False, values, alpha, beta)
            max_eval = max(max_eval, eval)
            alpha = max(alpha, eval)
            if beta <= alpha:
                break  # Beta cutoff
        return max_eval
    else:
        min_eval = math.inf
        for i in range(2):  
            eval = alpha_beta_pruning(depth - 1, node_index * 2 + i, True, values, alpha, beta)
            min_eval = min(min_eval, eval)
            beta = min(beta, eval)
            if beta <= alpha:
                break  
        return min_eval

if __name__ == "__main__":

    depth = 3  
    optimal_value = alpha_beta_pruning(depth, 0, True, values, -math.inf, math.inf)
    print(f"The optimal value is: {optimal_value}")

from collections import deque

def get_valid_moves(state):
    moves = []
    for i in range(3):
        if not state[i]:
            continue  
        disk = state[i][-1]  
        for j in range(3):
            if i == j:
                continue
            if not state[j] or state[j][-1] > disk:

                new_state = list(list(peg) for peg in state)
                new_state[i].pop()
                new_state[j].append(disk)
                moves.append(tuple(tuple(peg) for peg in new_state))
    return moves

def solve_hanoi_bfs(n):
    initial_state = (tuple(range(n, 0, -1)), (), ())
    goal_state = ((), (), tuple(range(n, 0, -1)))
    
    queue = deque([(initial_state, [initial_state])])
    visited = {initial_state}
    
    while queue:
        current_state, path = queue.popleft()
        
        if current_state == goal_state:
            return path
        
        for next_state in get_valid_moves(current_state):
            if next_state not in visited:
                visited.add(next_state)
                queue.append((next_state, path + [next_state]))
                
    return None

solution_path = solve_hanoi_bfs(3)
print(f"Total steps: {len(solution_path) - 1}")

    print(f"Step {step}: {st}")

def intelligent_guessing_agent(low=1, high=100):
    print(f"Think of a number between {low} and {high}, and I will find it!")
    print("Provide feedback: 'high' (if my guess is too high), "
          "'low' (if too low), or 'correct' (if I got it right).\n")
    
    attempts = 0
    current_low = low
    current_high = high

    while current_low <= current_high:
        guess = (current_low + current_high) // 2
        attempts += 1
        
        print(f"My guess is: {guess}")
        feedback = input("Is this high, low, or correct? ").strip().lower()
        
        if feedback == 'correct':
            print(f"Yay! I guessed your number in {attempts} attempts!")
            break
        elif feedback == 'high':
            current_high = guess - 1
        elif feedback == 'low':
            current_low = guess + 1
        else:
            print("Invalid feedback! Please type 'high', 'low', or 'correct'.")
            attempts -= 1  


if __name__ == "__main__":
    intelligent_guessing_agent(1, 100)

import random
import re

class RuleBasedChatbot:
    def __init__(self):

        self.rules = {
            r'\b(hi|hello|hey|hola)\b': [
                "Hello! How can I help you today?",
                "Hey there! What's on your mind?",
                "Hi! Good to chat with you."
            ],
            r'\b(how are you|how\'s it going|how do you do)\b': [
                "I'm just a bundle of code, but I'm running smoothly! How are you?",
                "Doing great, thanks for asking! What can I do for you?"
            ],
            r'\b(what is your name|who are you|your name)\b': [
                "I am EchoBot, your friendly neighborhood rule-based assistant!",
                "You can call me EchoBot."
            ],
            r'\b(help|support|features)\b': [
                "I can answer basic greetings, tell you my name, or help you exit the chat.",
                "Right now, I'm configured for basic conversational rules. Ask me who I am!"
            ],
            r'\b(weather)\b': [
                "I don't have real-time internet access, but I hope it's sunny where you are!",
                "I'm stuck inside a computer terminal, so I can't quite tell!"
            ]
        }
        

        self.exit_commands = ["bye", "goodbye", "exit", "quit"]
        

        self.default_responses = [
            "I'm not quite sure I understand that. Could you rephrase it?",
            "Interesting... Can you tell me more about that?",
            "My rules don't cover that query yet, but I'm listening!"
        ]

    def match_intent(self, user_message):
        user_message = user_message.lower().strip()
        

        if any(cmd in user_message for cmd in self.exit_commands):
            return "Goodbye! Have a wonderful day!"
            

        for pattern, responses in self.rules.items():
            if re.search(pattern, user_message):
                return random.choice(responses)
                

        return random.choice(self.default_responses)

    def start_chat(self):

        print("="*50)
        print(" EchoBot: Hello! Type 'exit', 'quit', or 'bye' to leave.")
        print("="*50)
        
        while True:
            try:
                user_input = input("You: ")
                response = self.match_intent(user_input)
                print(f"EchoBot: {response}")
                
                if response == "Goodbye! Have a wonderful day!":
                    break
            except (KeyboardInterrupt, EOFError):
                print("\nEchoBot: Goodbye!")
                break


if __name__ == "__main__":
    bot = RuleBasedChatbot()
    bot.start_chat()

import math
import random

HUMAN = 'O'
AI = 'X'
EMPTY = ' '

def print_board(board):
    print("\n" + "  +---+---+---+")
    for row in board:
        print("  | " + " | ".join(row) + " |")
        print("  +---+---+---+")
    print()

def check_winner(board):

    for i in range(3):
        if board[i][0] == board[i][1] == board[i][2] != EMPTY:
            return 10 if board[i][0] == AI else -10
        if board[0][i] == board[1][i] == board[2][i] != EMPTY:
            return 10 if board[0][i] == AI else -10
            
    if board[0][0] == board[1][1] == board[2][2] != EMPTY:
        return 10 if board[0][0] == AI else -10
    if board[0][2] == board[1][1] == board[2][0] != EMPTY:
        return 10 if board[0][2] == AI else -10

    return 0

def get_empty_cells(board):
    cells = []
    for r in range(3):
        for c in range(3):
            if board[r][c] == EMPTY:
                cells.append((r, c))
    return cells

def is_board_full(board):
    return len(get_empty_cells(board)) == 0

def minimax(board, depth, is_maximizing, alpha, beta):
    score = check_winner(board)

    if score == 10:
        return score - depth  
    if score == -10:
        return score + depth  
    if is_board_full(board):
        return 0

    if is_maximizing:
        max_eval = -math.inf
        for row, col in get_empty_cells(board):
            board[row][col] = AI
            evaluation = minimax(board, depth + 1, False, alpha, beta)
            board[row][col] = EMPTY  
            max_eval = max(max_eval, evaluation)
            alpha = max(alpha, evaluation)
            if beta <= alpha:
                break  
        return max_eval
    else:
        min_eval = math.inf
        for row, col in get_empty_cells(board):
            board[row][col] = HUMAN
            evaluation = minimax(board, depth + 1, True, alpha, beta)
            board[row][col] = EMPTY  
            min_eval = min(min_eval, evaluation)
            beta = min(beta, evaluation)
            if beta <= alpha:
                break  
        return min_eval

def find_best_move(board):
    best_value = -math.inf
    best_move = (-1, -1)

    for row, col in get_empty_cells(board):
        board[row][col] = AI

        move_value = minimax(board, 0, False, -math.inf, math.inf)
        board[row][col] = EMPTY  

        if move_value > best_value:
            best_value = move_value
            best_move = (row, col)

    return best_move

def main():

    board = [[EMPTY for _ in range(3)] for _ in range(3)]
    
    print("Welcome to Tic-Tac-Toe! You are 'O' and the AI is 'X'.")
    print_board(board)

    while True:
        
        empty_cells = get_empty_cells(board)
        if not empty_cells or check_winner(board) != 0:
            break
            
        try:
            move = int(input("Enter your move (1-9): ")) - 1
            row, col = move // 3, move % 3
            if (row, col) not in empty_cells:
                print("Invalid move! That cell is taken. Try again.")
                continue
            board[row][col] = HUMAN
        except (ValueError, IndexError):
            print("Invalid input! Please enter a number between 1 and 9.")
            continue

        print_board(board)

        if check_winner(board) == -10:
            print("Congratulations! You won!")
            break
        if is_board_full(board):
            print("It's a tie!")
            break

        print("AI is calculating the optimal move...")
        ai_row, ai_col = find_best_move(board)
        board[ai_row][ai_col] = AI
        print_board(board)


        if check_winner(board) == 10:
            print("The AI wins! Better luck next time.")
            break
        if is_board_full(board):
            print("It's a tie!")
            break

if __name__ == "__main__":
    main()




BACKTRACKING:


def solve_8_queens():
    N = 8
    

    cols = set()        
    diag1 = set()       
    diag2 = set()       
    
    
    current_assignment = [-1] * N
    all_solutions = []

    def backtrack(row):

        if row == N:
            all_solutions.append(list(current_assignment))
            return

        for col in range(N):
            if col in cols or (row + col) in diag1 or (row - col) in diag2:
                continue  

            cols.add(col)
            diag1.add(row + col)
            diag2.add(row - col)
            current_assignment[row] = col

            backtrack(row + 1)

            cols.remove(col)
            diag1.remove(row + col)
            diag2.remove(row - col)
            current_assignment[row] = -1

    backtrack(0)
    return all_solutions

def print_board(solution):
    for col in solution:
        row_str = [" . "] * 8
        row_str[col] = " Q "
        print("".join(row_str))
    print("\n" + "="*24 + "\n")


solutions = solve_8_queens()

print(f"Total valid configurations found: {len(solutions)}\n")
print("First configuration matrix layout:")
